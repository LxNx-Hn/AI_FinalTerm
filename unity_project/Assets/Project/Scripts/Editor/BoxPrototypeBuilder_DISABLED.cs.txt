    [MenuItem("Project Tools/Build Box Prototype (All)")]
    public static void BuildAll()
    {
        AssetDatabase.Refresh();

        // Ensure generated assets exist
        if (!File.Exists(S_ENTRY)) SceneBootstrapEditor.BootstrapMissingScenes();
        if (!File.Exists(T_FLOOR_A))
            FloorTileCreator.CreateFloorTilePlaceholders();

        // Ensure placeholder sprites are assigned to all prefabs
        PlaceholderSpriteCreator.CreateAll();
        EnsureAttackHitboxPrefab();
        // After placeholders, overlay any imported real assets (Evelyn idle, item icons).
        // Order matters: ImportedAssetApplier wins so BuildAll never reverts real art.
        ImportedAssetApplier.ApplyAll();
        AssetDatabase.Refresh();

        BuildEntry();
        BuildStage02();
        BuildBoss01();
        BuildRooftopEndingWalk();
        BuildEndingCredits();

        AssetDatabase.SaveAssets();
        AssetDatabase.Refresh();
        Debug.Log("[BoxPrototype] ✅ All scenes built.");
    }

    [MenuItem("Project Tools/Boss/Create Boss Animator Controller")]
    public static void CreateBossAnimatorController()
    {
        const string ANIM_BASE = "Assets/Project/Resources/Boss/BOSSASSET/BOSS/ANIME/";
        const string CTRL_PATH = ANIM_BASE + "BossAnimator.controller";

        // 상태 목록 — DISAPPEAR_* 제외, 29개 상태 + MARKATK = 29 상태
        var stateNames = new string[]
        {
            "IDLE_DOWN",  "IDLE_UP",  "IDLE_LEFT",  "IDLE_RIGHT",
            "MOVE_DOWN",  "MOVE_UP",  "MOVE_LEFT",  "MOVE_RIGHT",
            "ATTACK_DOWN","ATTACK_UP","ATTACK_LEFT", "ATTACK_RIGHT",
            "ATTACK_READY_DOWN","ATTACK_READY_UP","ATTACK_READY_LEFT","ATTACK_READY_RIGHT",
            "SCREAM_DOWN","SCREAM_UP","SCREAM_LEFT", "SCREAM_RIGHT",
            "APPEAR_DOWN","APPEAR_UP","APPEAR_LEFT", "APPEAR_RIGHT",
            "DIE_DOWN",   "DIE_UP",   "DIE_LEFT",    "DIE_RIGHT",
            "MARKATK"
        };

        AnimatorController ctrl;
        if (File.Exists(CTRL_PATH))
        {
            ctrl = AssetDatabase.LoadAssetAtPath<AnimatorController>(CTRL_PATH);
            if (ctrl == null)
            {
                Debug.LogError($"[BossAnim] Failed to load existing controller at {CTRL_PATH}");
                return;
            }
        }
        else
        {
            ctrl = AnimatorController.CreateAnimatorControllerAtPath(CTRL_PATH);
        }

        var sm = ctrl.layers[0].stateMachine;

        foreach (string stateName in stateNames)
        {
            // 이미 존재하는 상태는 건너뜀
            bool exists = false;
            foreach (var s in sm.states)
                if (s.state.name == stateName) { exists = true; break; }
            if (exists) continue;

            string clipPath = ANIM_BASE + stateName + ".anim";
            var clip = AssetDatabase.LoadAssetAtPath<AnimationClip>(clipPath);

            var state = sm.AddState(stateName);
            if (clip != null) state.motion = clip;
            else Debug.LogWarning($"[BossAnim] Clip not found: {clipPath}");
        }

        // 기본 상태를 IDLE_DOWN으로 설정
        foreach (var cs in sm.states)
            if (cs.state.name == "IDLE_DOWN")
            {
                sm.defaultState = cs.state;
                break;
            }

        EditorUtility.SetDirty(ctrl);
        AssetDatabase.SaveAssets();
        AssetDatabase.Refresh();
        Debug.Log($"[BossAnim] ✅ BossAnimator.controller created/updated at {CTRL_PATH}");
    }

    static void BuildScene(string path, System.Action setup)
    {
        if (!File.Exists(path))
        {
            Debug.LogWarning($"[BoxPrototype] Scene not found: {path}. Run Bootstrap first.");
            return;
        }

        var scene = EditorSceneManager.OpenScene(path, OpenSceneMode.Single);
        setup();
        EditorSceneManager.SaveScene(scene);
        Debug.Log($"[BoxPrototype] Saved: {path}");
    }

    static void SetupEntry()
    {
        EnsureCamera(null, 5f);
        EnsureEventSystem();

        var sys = FindOrCreate("00_System");
        var loader = EnsureComponent<SceneLoader>(sys);

        // Canvas
        var canvas = GetOrCreateCanvas("Canvas");
        var ct = canvas.transform;

        var bg = MakeUIPanel(ct, "BackgroundPanel", Color.black);
        StretchFull(bg);

        var title = MakeUIText(ct, "TitleText", "CODE BLUE", 72, TextAlignmentOptions.Center);
        SetRect(title, new Vector2(0.1f, 0.6f), new Vector2(0.9f, 0.85f));

        var startBtn = MakeButton(ct, "StartButton", "START", loader, "Stage01_ParkingToHospital");
        SetRect(startBtn, new Vector2(0.35f, 0.42f), new Vector2(0.65f, 0.55f));

        var quitBtn = MakeQuitButton(ct, "QuitButton", "QUIT", loader);
        SetRect(quitBtn, new Vector2(0.35f, 0.26f), new Vector2(0.65f, 0.39f));

        MakeReadme("00_System", "Entry: START→Stage01, QUIT→quit. Edit TitleText, button text, background color.");
    }

    // =========================================================================
    // Boss01 — Elevator
    // =========================================================================
    static void SetupBoss01()
    {
        var sys  = FindOrCreate("00_System");
        var ui   = FindOrCreate("07_UI");
        var cam  = FindOrCreate("08_Camera");

        PlacePrefab(P_GRID, Vector3.zero, sys.transform, "GridManager");
        EnsureEventSystem(sys.transform);

        // vFinal-Lock-4 시작 위치: 플레이어 (0, -2), 보스 (0, 2)
        var player = PlacePrefab(P_PLAYER, new Vector3(0, -2, 0), null, "Player");
        WireAttackHitbox(player);
        WirePlayerAnimator(player);

        var camGo = EnsureCamera(cam.transform, 5f);
        var cf = EnsureComponent<CameraFollow>(camGo);
        cf.target = player.transform;
        cf.smoothSpeed = 8f;
        cf.offset = new Vector3(0, 0, -10);

        // ── Boss object ────────────────────────────────────────────────────────
        var bossGo = FindOrCreateWithComponent<BossHealth>("Boss_Elevator");
        bossGo.transform.position = new Vector3(0, 2, 0); // 명세 시작 위치
        EnsureComponent<BossPatternCaster>(bossGo);
        var bc = EnsureComponent<ElevatorBossController>(bossGo);

        // Non-blocking occupant — player can walk through boss; PlayerCombat hits via BossCell property
        var bossOcc = EnsureComponent<GridOccupant>(bossGo);
        bossOcc.blocksMovement = false;

        // Boss sprite (visible placeholder, 64×64 purple)
        var bossSpriteGo = FindOrCreateChild(bossGo.transform, "BossSprite");
        bossSpriteGo.transform.localPosition = Vector3.zero;
        var bossSR = EnsureComponent<SpriteRenderer>(bossSpriteGo);
        var bossPH = PlaceholderSpriteCreator.GetSprite("Boss_PH");
        if (bossPH != null)
        {
            bossSR.sprite           = bossPH;
            bossSR.sortingLayerName = "Boss";
            bossSR.sortingOrder     = 0;
        }

        // Animator — IDLE_DOWN.anim 등 클립의 path: "" 바인딩 때문에
        //            Animator는 SpriteRenderer와 같은 오브젝트(BossSprite)에 있어야 한다.
        //            Root(Boss_Elevator)에 두면 clip이 SpriteRenderer를 찾지 못함.
        var bossAnimator = EnsureComponent<Animator>(bossSpriteGo);
        const string BOSS_CTRL_PATH = "Assets/Project/Resources/Boss/BOSSASSET/BOSS/ANIME/BossAnimator.controller";
        var animCtrl = AssetDatabase.LoadAssetAtPath<RuntimeAnimatorController>(BOSS_CTRL_PATH);
        if (animCtrl != null)
            bossAnimator.runtimeAnimatorController = animCtrl;
        else
            Debug.LogWarning("[Boss] BossAnimator.controller not found. Run 'Project Tools > Boss > Create Boss Animator Controller' first.");

        // BossAnimatorDriver — animator + bossSpriteRenderer 와이어링 (드라이버는 Root에 유지)
        var animDriver = EnsureComponent<BossAnimatorDriver>(bossGo);
        var animDriverSO = new SerializedObject(animDriver);
        animDriverSO.FindProperty("animator").objectReferenceValue = bossAnimator;
        animDriverSO.FindProperty("bossSpriteRenderer").objectReferenceValue = bossSR;
        animDriverSO.ApplyModifiedProperties();

        // AudioSource — 2D, playOnAwake=false
        var bossAudio = EnsureComponent<AudioSource>(bossGo);
        bossAudio.playOnAwake  = false;
        bossAudio.spatialBlend = 0f;

        // PatternOrigin — standalone at arena center (NOT child of boss, so boss teleports don't shift it)
        var originGo = GameObject.Find("PatternOrigin");
        if (originGo == null) originGo = new GameObject("PatternOrigin");
        originGo.transform.SetParent(sys.transform, false);
        originGo.transform.position = Vector3.zero;

        // Arena floor (7x7 centered)
        var tm = EnsureFloorTilemap();
        FillFloor(tm, -3, -3, 3, 3);

        // Arena outer walls
        var walls = FindOrCreate("02_Walls");
        FillWallRow(walls.transform, -4, 4, -4, "Wall_Boss01_Bottom");
        FillWallRow(walls.transform, -4, 4,  4, "Wall_Boss01_Top");
        FillWallCol(walls.transform,  4, -4, -4, "Wall_Boss01_Left");
        FillWallCol(walls.transform,  4,  4,  4, "Wall_Boss01_Right");

        // Wire BossPatternCaster references via serialized fields
        var casterSO = new SerializedObject(bossGo.GetComponent<BossPatternCaster>());
        casterSO.FindProperty("patternOrigin").objectReferenceValue = originGo.transform;
        var dmgPfb = AssetDatabase.LoadAssetAtPath<GameObject>(P_DMG);
        var wrnPfb = AssetDatabase.LoadAssetAtPath<GameObject>(P_WARN);
        var sfePfb = AssetDatabase.LoadAssetAtPath<GameObject>(P_SAFE);
        if (dmgPfb) casterSO.FindProperty("damageTilePrefab").objectReferenceValue = dmgPfb;
        if (wrnPfb) casterSO.FindProperty("warningTilePrefab").objectReferenceValue = wrnPfb;
        if (sfePfb) casterSO.FindProperty("safeTilePrefab").objectReferenceValue = sfePfb;
        casterSO.ApplyModifiedProperties();

        // Wire ElevatorBossController
        var ctrlSO = new SerializedObject(bc);
        ctrlSO.FindProperty("nextSceneName").stringValue = "RooftopEndingWalk";
        AssignObjectReferenceIfNull(ctrlSO, "bossSpriteTransform", bossSpriteGo.transform);
        // 애니메이션 방향별 클립 사용 — 회전 불필요
        AssignBoolIfDefault(ctrlSO, "rotateBossVisualByFacing", false);
        AssignFloatIfZero(ctrlSO, "visualRotationOffsetDegrees", 0f);
        // bossSpriteYOffset: IDLE_DOWN.anim 첫 프레임 sprite로 자동 계산
        float autoYOffset = CalcSpriteHalfHeight(
            "Assets/Project/Resources/Boss/BOSSASSET/BOSS/ANIME/IDLE_DOWN.anim");
        AssignFloatIfZero(ctrlSO, "bossSpriteYOffset", autoYOffset);
        var bhSO = bossGo.GetComponent<BossHealth>();
        AssignObjectReferenceIfNull(ctrlSO, "bossHealth", bhSO);
        AssignObjectReferenceIfNull(ctrlSO, "caster", bossGo.GetComponent<BossPatternCaster>());
        AssignObjectReferenceIfNull(ctrlSO, "animDriver", animDriver);
        AssignObjectReferenceIfNull(ctrlSO, "bossAudioSource", bossAudio);

        // VFX 프리팹 와이어링
        var scratchPfb = AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Project/Prefabs/Boss/ScratchVFX.prefab");
        var slamPfb    = AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Project/Prefabs/Boss/SlamVFX.prefab");
        var smokePfb   = AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Project/Prefabs/Boss/DashSmokeVFX.prefab");
        var markAtkPfb = AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Project/Prefabs/Boss/MarkATKVFX.prefab");
        if (scratchPfb != null) AssignObjectReferenceIfNull(ctrlSO, "scratchVFXPrefab", scratchPfb);
        if (slamPfb    != null) AssignObjectReferenceIfNull(ctrlSO, "slamVFXPrefab", slamPfb);
        if (smokePfb   != null) AssignObjectReferenceIfNull(ctrlSO, "dashSmokeVFXPrefab", smokePfb);
        if (markAtkPfb != null) AssignObjectReferenceIfNull(ctrlSO, "markAtkVFXPrefab", markAtkPfb);

        ctrlSO.ApplyModifiedProperties();

        // ── Boss UI ────────────────────────────────────────────────────────────
        var canvas = GetOrCreateCanvas("Canvas");
        var ct = canvas.transform;

        // Remove legacy flat HPText at canvas root from old builds
        var legacyHp = ct.Find("HPText");
        if (legacyHp != null) Object.DestroyImmediate(legacyHp.gameObject);

        // Player HP panel (background + text)
        var hpPanel = FindOrCreateUIChild(ct, "HPPanel");
        SetRect(hpPanel, new Vector2(0f, 0.88f), new Vector2(0.28f, 1f));
        var hpBg = EnsureComponent<Image>(hpPanel);
        hpBg.color = new Color(0f, 0f, 0f, 0.6f);
        var hpTextGo = FindOrCreateUIChild(hpPanel.transform, "HPText");
        var hpText = EnsureComponent<TextMeshProUGUI>(hpTextGo);
        hpText.text = "HP: 3 / 3";
        hpText.fontSize = 36;
        hpText.color = Color.white;
        hpText.alignment = TextAlignmentOptions.MidlineLeft;
        SetRect(hpTextGo, new Vector2(0.05f, 0f), new Vector2(0.95f, 1f));
        var hpUI = EnsureComponent<HpTextUI>(hpTextGo);
        var hpUISO = new SerializedObject(hpUI);
        hpUISO.FindProperty("hpText").objectReferenceValue = hpText;
        hpUISO.FindProperty("heartRoot").objectReferenceValue = hpPanel.GetComponent<RectTransform>();
        hpUISO.FindProperty("heartContainerPrefab").objectReferenceValue =
            AssetDatabase.LoadAssetAtPath<GameObject>("Assets/HeartSystem/Prefabs/HeartContainer.prefab");
        hpUISO.ApplyModifiedProperties();

        // Boss HP panel (background + text)
        var bossPanel = FindOrCreateUIChild(ct, "BossHPPanel");
        SetRect(bossPanel, new Vector2(0.28f, 0.88f), new Vector2(0.72f, 1f));
        var bossPanelBg = EnsureComponent<Image>(bossPanel);
        bossPanelBg.color = new Color(0f, 0f, 0f, 0.6f);
        var bossTextGo = FindOrCreateUIChild(bossPanel.transform, "BossHPText");
        var bossText = EnsureComponent<TextMeshProUGUI>(bossTextGo);
        bossText.text = "BOSS: 40 / 40";
        bossText.fontSize = 36;
        bossText.color = new Color(1f, 0.3f, 0.3f, 1f);
        bossText.alignment = TextAlignmentOptions.Center;
        SetRect(bossTextGo, new Vector2(0.05f, 0f), new Vector2(0.95f, 1f));
        var bossUI = EnsureComponent<BossHpTextUI>(bossPanel);
        var bossUISO = new SerializedObject(bossUI);
        bossUISO.FindProperty("hpText").objectReferenceValue = bossText;
        bossUISO.FindProperty("bossHealth").objectReferenceValue = bhSO;

    static GameObject PlacePrefab(string path, Vector3 pos, Transform parent, string name)
    {
        var pfb = AssetDatabase.LoadAssetAtPath<GameObject>(path);
        if (pfb == null)
        {
            Debug.LogWarning($"[BoxPrototype] Prefab not found: {path}");
            var empty = new GameObject(name);
            empty.transform.SetParent(parent, false);
            empty.transform.position = pos;
            return empty;
        }

        // Check if already exists in scene with this name — if so, force-update position
        // so SetupXxx changes to start positions take effect on rebuild.
        if (parent != null)
        {
            var existing = parent.Find(name);
            if (existing != null) { existing.position = pos; return existing.gameObject; }
        }
        else
        {
            var found = GameObject.Find(name);
            if (found != null) { found.transform.position = pos; return found; }
        }

        var go = (GameObject)PrefabUtility.InstantiatePrefab(pfb);
        go.name = name;
        go.transform.position = pos;
        if (parent != null) go.transform.SetParent(parent, true);
        return go;
    }

    static GameObject PlaceEnemy(string path, int x, int y, Transform parent, string name)

            so.FindProperty("startActive").boolValue = false;
            so.ApplyModifiedProperties();
        }
        go.SetActive(true);
        return go;
    }

    static Tilemap EnsureFloorTilemap()
    {
        var gridGo = GameObject.Find("01_Floor");
        if (gridGo == null)
            gridGo = new GameObject("01_Floor");

        // If no Tilemap exists yet, clear legacy individual tile GameObjects
        if (gridGo.GetComponentInChildren<Tilemap>() == null)
        {
            for (int i = gridGo.transform.childCount - 1; i >= 0; i--)
                Object.DestroyImmediate(gridGo.transform.GetChild(i).gameObject);
        }

        var grid = gridGo.GetComponent<Grid>();
        if (grid == null) grid = gridGo.AddComponent<Grid>();
        grid.cellSize = new Vector3(1, 1, 0);

        var tilemapGo = gridGo.transform.Find("FloorMap")?.gameObject;
        if (tilemapGo == null)
        {
            tilemapGo = new GameObject("FloorMap");
            tilemapGo.transform.SetParent(gridGo.transform, false);
        }

        var tm = tilemapGo.GetComponent<Tilemap>();
        if (tm == null) tm = tilemapGo.AddComponent<Tilemap>();

        var tmr = tilemapGo.GetComponent<TilemapRenderer>();
        if (tmr == null) tmr = tilemapGo.AddComponent<TilemapRenderer>();
        tmr.sortingLayerName = "Floor";
        tmr.sortingOrder = 0;

        return tm;
    }

    static void FillFloor(Tilemap tm, int xMin, int yMin, int xMax, int yMax)
    {
        var tileA = AssetDatabase.LoadAssetAtPath<TileBase>(T_FLOOR_A);
        var tileB = AssetDatabase.LoadAssetAtPath<TileBase>(T_FLOOR_B);
        if (tileA == null || tileB == null)
        {
            Debug.LogWarning("[BoxPrototype] FloorTile .asset files missing. Run 'Create FloorTile Placeholders' first.");
            return;
        }
        for (int x = xMin; x <= xMax; x++)
            for (int y = yMin; y <= yMax; y++)
                tm.SetTile(new Vector3Int(x, y, 0), (x + y) % 2 == 0 ? tileA : tileB);
    }

    static Canvas GetOrCreateCanvas(string name)
    {
        var existing = Object.FindObjectOfType<Canvas>();
        if (existing != null && existing.name == name) return existing;

        // Canvas needs RectTransform — create with all UI components in one go
        // so the Transform is RectTransform from the start.
        var go = GameObject.Find(name);
        if (go == null)
        {
            go = new GameObject(name,
                typeof(RectTransform),
                typeof(Canvas),
                typeof(CanvasScaler),
                typeof(GraphicRaycaster));
        }
        var canvas = EnsureComponent<Canvas>(go);
        canvas.renderMode = RenderMode.ScreenSpaceOverlay;
        canvas.sortingOrder = 100;
        var scaler = EnsureComponent<CanvasScaler>(go);
        scaler.uiScaleMode         = CanvasScaler.ScaleMode.ScaleWithScreenSize;
        scaler.referenceResolution = new Vector2(960, 540);
        scaler.matchWidthOrHeight  = 0f;
        EnsureComponent<GraphicRaycaster>(go);
        return canvas;
    }

    // UI children must have RectTransform. `new GameObject(name)` creates a plain
    // Transform; AddComponent<RectTransform> on it later is a no-op. Force the
    // RectTransform at construction time.
    static GameObject FindOrCreateUIChild(Transform parent, string name)
    {
        var existing = parent.Find(name);
        if (existing != null) return existing.gameObject;
        var go = new GameObject(name, typeof(RectTransform));
        go.transform.SetParent(parent, false);
        return go;
    }

    // Generates Assets/Project/Prefabs/Pattern/AttackHitbox.prefab if missing,
    // and wires its sprite to the AttackHitbox_PH placeholder.
    static void EnsureAttackHitboxPrefab()
    {
        var sprite = PlaceholderSpriteCreator.GetSprite("AttackHitbox_PH");
        if (sprite == null)
        {
            Debug.LogWarning("[BoxPrototype] AttackHitbox_PH sprite missing — run PlaceholderSpriteCreator first.");
            return;
        }

        Directory.CreateDirectory(Path.Combine(Application.dataPath, "Project/Prefabs/Pattern"));

        if (!File.Exists(P_HITBOX))
        {
            var template = new GameObject("AttackHitbox");
            var sr = template.AddComponent<SpriteRenderer>();
            sr.sprite           = sprite;
            sr.sortingLayerName = "PatternWarning";
            sr.sortingOrder     = 5;
            PrefabUtility.SaveAsPrefabAsset(template, P_HITBOX);
            Object.DestroyImmediate(template);
        }
        else
        {
            using var scope = new PrefabUtility.EditPrefabContentsScope(P_HITBOX);
            var sr = scope.prefabContentsRoot.GetComponent<SpriteRenderer>();
            if (sr == null) sr = scope.prefabContentsRoot.AddComponent<SpriteRenderer>();
            sr.sprite           = sprite;
            sr.sortingLayerName = "PatternWarning";
            sr.sortingOrder     = 5;
        }
    }

    // Adds PlayerVisualAnimator2D and wires all sprite arrays from Stage01_Parking player art.
    static void WirePlayerAnimator(GameObject player)
    {
        if (player == null) return;
        var anim = EnsureComponent<PlayerVisualAnimator2D>(player);
        var so = new SerializedObject(anim);

        const string BASE = "Assets/Project/Art/Stage01_Parking/Player/";

        Sprite[] Load(string tex, params string[] names)
        {
            var all = AssetDatabase.LoadAllAssetsAtPath(tex);
            var result = new List<Sprite>();
            foreach (var n in names)
                foreach (var a in all)

            return result.ToArray();

    static void SetSpriteArray(SerializedObject so, string fieldName, Sprite[] sprites)
    {
        var prop = so.FindProperty(fieldName);
        if (prop == null) return;
        prop.arraySize = sprites.Length;
        for (int i = 0; i < sprites.Length; i++)
            prop.GetArrayElementAtIndex(i).objectReferenceValue = sprites[i];
    }

    static GameObject FindOrCreateChild(Transform parent, string name)
    {
        var existing = parent.Find(name);
        if (existing != null) return existing.gameObject;
        var go = new GameObject(name);
        go.transform.SetParent(parent, false);
        return go;
    }

    static GameObject FindOrCreateWithComponent<T>(string name) where T : Component
    {
        var found = Object.FindObjectOfType<T>();
        if (found != null) return found.gameObject;
        var go = new GameObject(name);
        go.AddComponent<T>();
        return go;
    }

    static T EnsureComponent<T>(GameObject go) where T : Component
    {
        var c = go.GetComponent<T>();
        return c != null ? c : go.AddComponent<T>();
    }

    static GameObject EnsureCamera(Transform parent, float orthoSize)
    {
        var camGo = GameObject.Find("Main Camera");
        if (camGo == null)
        {
            camGo = new GameObject("Main Camera");
            camGo.tag = "MainCamera";
        }
        if (parent != null && camGo.transform.parent != parent)
            camGo.transform.SetParent(parent, false);

        var cam = EnsureComponent<Camera>(camGo);
        cam.orthographic = true;
        cam.orthographicSize = orthoSize;
        cam.clearFlags = CameraClearFlags.SolidColor;
        cam.backgroundColor = new Color(0.05f, 0.05f, 0.1f);
        camGo.transform.position = new Vector3(0, 0, -10);
        return camGo;
    }

    static void EnsureEventSystem(Transform parent = null)
    {
        if (Object.FindObjectOfType<EventSystem>() != null) return;
        var esGo = new GameObject("EventSystem");
        if (parent != null) esGo.transform.SetParent(parent, false);
        esGo.AddComponent<EventSystem>();
        esGo.AddComponent<StandaloneInputModule>();
    }

    static void SetGridItemType(GameObject go, ItemType type)
    {
        var item = go.GetComponent<GridItem>();
        if (item == null) return;
        var so = new SerializedObject(item);
        so.FindProperty("itemType").enumValueIndex = (int)type;
        so.ApplyModifiedProperties();
    }

    static void MakeReadme(string parentName, string content)
    {
        var parent = GameObject.Find(parentName);
        Transform p = parent != null ? parent.transform : null;
        var existing = p != null ? p.Find("README_SCENE_GUIDE") : null;
        if (existing != null) Object.DestroyImmediate(existing.gameObject);

        var go = new GameObject("README_SCENE_GUIDE [" + content.Substring(0, Mathf.Min(40, content.Length)) + "...]");
        if (p != null) go.transform.SetParent(p, false);
    }

    static void AssignObjectReferenceIfNull(SerializedObject so, string propertyName, Object value)
    {
        var prop = so.FindProperty(propertyName);
        if (prop != null && prop.objectReferenceValue == null)
            prop.objectReferenceValue = value;
    }

    static void AssignFloatIfZero(SerializedObject so, string propertyName, float value)
    {
        var prop = so.FindProperty(propertyName);
        if (prop != null && Mathf.Approximately(prop.floatValue, 0f))
            prop.floatValue = value;
    }

    static void AssignBoolIfDefault(SerializedObject so, string propertyName, bool value)
    {
        var prop = so.FindProperty(propertyName);
        if (prop != null && prop.boolValue == default(bool))
            prop.boolValue = value;
    }

    /// <summary>IDLE_DOWN.anim 등의 첫 프레임 sprite에서 world 단위 반높이를 반환합니다.</summary>
    static float CalcSpriteHalfHeight(string animPath)
    {
        var clip = AssetDatabase.LoadAssetAtPath<AnimationClip>(animPath);
        if (clip == null) return 0f;

        var bindings = AnimationUtility.GetObjectReferenceCurveBindings(clip);
        foreach (var b in bindings)
        {
            if (b.propertyName != "m_Sprite") continue;
            var keys = AnimationUtility.GetObjectReferenceCurve(clip, b);
            if (keys.Length == 0) continue;
            if (keys[0].value is Sprite spr)
                return spr.bounds.extents.y;
        }
        return 0f;
    }

    static void EnsureSimpleVFXPrefab(string prefabPath, string spritePath,
        string sortingLayer, int sortingOrder, bool forceRebuild)
    {
        var sprite = AssetDatabase.LoadAssetAtPath<Sprite>(spritePath);

        if (File.Exists(prefabPath) && !forceRebuild)
        {
            return;
        }

        if (File.Exists(prefabPath))
        {
            using var scope = new PrefabUtility.EditPrefabContentsScope(prefabPath);
            var sr = scope.prefabContentsRoot.GetComponent<SpriteRenderer>();
            if (sr == null) sr = scope.prefabContentsRoot.AddComponent<SpriteRenderer>();
            if (sprite != null) sr.sprite = sprite;
            sr.sortingLayerName = sortingLayer;
            sr.sortingOrder     = sortingOrder;
        }
        else
        {
            var tmpl = new GameObject(Path.GetFileNameWithoutExtension(prefabPath));
            var sr = tmpl.AddComponent<SpriteRenderer>();
            if (sprite != null) sr.sprite = sprite;
            sr.sortingLayerName = sortingLayer;
            sr.sortingOrder     = sortingOrder;
            PrefabUtility.SaveAsPrefabAsset(tmpl, prefabPath);
            Object.DestroyImmediate(tmpl);
        }
    }

    static void EnsureAnimatedVFXPrefab(string prefabPath, string spritePath, string animPath,
        string sortingLayer, int sortingOrder, bool forceRebuild)
    {
        var sprite = AssetDatabase.LoadAssetAtPath<Sprite>(spritePath);
        var clip   = AssetDatabase.LoadAssetAtPath<AnimationClip>(animPath);

        // Build or reuse a standalone AnimatorController for this VFX
        string ctrlPath = Path.ChangeExtension(prefabPath, null) + "_Ctrl.controller";
        AnimatorController ctrl;

        if (!File.Exists(ctrlPath))
        {
            ctrl = AnimatorController.CreateAnimatorControllerAtPath(ctrlPath);
            if (clip != null)
            {
                var state = ctrl.layers[0].stateMachine.AddState("Play");
                state.motion = clip;
                ctrl.layers[0].stateMachine.defaultState = state;
            }
        }
        else
        {
            ctrl = AssetDatabase.LoadAssetAtPath<AnimatorController>(ctrlPath);
        }

        if (File.Exists(prefabPath) && !forceRebuild)
        {
            return;
        }

        if (File.Exists(prefabPath))
        {
            using var scope = new PrefabUtility.EditPrefabContentsScope(prefabPath);
            var root = scope.prefabContentsRoot;
            var sr = root.GetComponent<SpriteRenderer>();
            if (sr == null) sr = root.AddComponent<SpriteRenderer>();
            if (sprite != null) sr.sprite = sprite;
            sr.sortingLayerName = sortingLayer;
            sr.sortingOrder     = sortingOrder;

            var anim = root.GetComponent<Animator>();
            if (anim == null) anim = root.AddComponent<Animator>();
            if (ctrl != null) anim.runtimeAnimatorController = ctrl;
        }
        else
        {
            var tmpl = new GameObject(Path.GetFileNameWithoutExtension(prefabPath));
            var sr = tmpl.AddComponent<SpriteRenderer>();
            if (sprite != null) sr.sprite = sprite;
            sr.sortingLayerName = sortingLayer;
            sr.sortingOrder     = sortingOrder;

            var anim = tmpl.AddComponent<Animator>();
            if (ctrl != null) anim.runtimeAnimatorController = ctrl;

            PrefabUtility.SaveAsPrefabAsset(tmpl, prefabPath);
            Object.DestroyImmediate(tmpl);
        }
    }
