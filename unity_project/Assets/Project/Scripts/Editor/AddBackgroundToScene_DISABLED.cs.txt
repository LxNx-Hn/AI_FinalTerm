using UnityEditor; using UnityEditor.SceneManagement; using UnityEngine; public class AddBackgroundToScene { [MenuItem("Tools/Add Boss Background")] public static void AddBg() { var scene = EditorSceneManager.OpenScene("Assets/Project/Scenes/Boss01_Elevator.unity"); var existing = Object.FindObjectOfType<ElevatorLoopBackgroundRuntime>(); if (existing != null) { Object.DestroyImmediate(existing.gameObject); } var go = new GameObject("ElevatorLoopBackground"); go.AddComponent<ElevatorLoopBackgroundRuntime>(); EditorSceneManager.SaveScene(scene); Debug.Log("Background added to scene and saved."); } }
