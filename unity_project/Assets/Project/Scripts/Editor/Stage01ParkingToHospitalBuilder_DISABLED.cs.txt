    private static void ConfigureExistingSpriteSheet(string assetPath, int columns, int rows, float ppu, SpriteAlignment alignment)
    {
        string absolutePath = ToAbsoluteAssetPath(assetPath);
        if (!File.Exists(absolutePath))
        {
            return;
        }

        AssetDatabase.ImportAsset(assetPath, ImportAssetOptions.ForceUpdate);
        TextureImporter importer = AssetImporter.GetAtPath(assetPath) as TextureImporter;
        if (importer == null)
        {
            return;
        }

        Texture2D temp = LoadTextureFromFile(absolutePath);
        int frameWidth = temp.width / columns;
        int frameHeight = temp.height / rows;
        int textureHeight = temp.height;

        importer.textureType = TextureImporterType.Sprite;
        importer.spriteImportMode = SpriteImportMode.Multiple;
        importer.spritePixelsPerUnit = ppu;
        importer.filterMode = FilterMode.Point;
        importer.textureCompression = TextureImporterCompression.Uncompressed;
        importer.alphaIsTransparency = true;

        var sprites = new List<SpriteMetaData>();
        for (int row = 0; row < rows; row++)
        {
            for (int col = 0; col < columns; col++)
            {
                sprites.Add(new SpriteMetaData
                {
                    alignment = (int)alignment,
                    pivot = alignment == SpriteAlignment.BottomCenter ? new Vector2(0.5f, 0f) : new Vector2(0.5f, 0.5f),
                    name = $"{Path.GetFileNameWithoutExtension(assetPath)}_r{row + 1:D2}_c{col + 1:D2}",
                    rect = new Rect(col * frameWidth, textureHeight - ((row + 1) * frameHeight), frameWidth, frameHeight)
                });
            }
        }

        importer.spritesheet = sprites.ToArray();
        importer.SaveAndReimport();
        UnityEngine.Object.DestroyImmediate(temp);
    }

    private static void ConfigureExistingSingleSprite(string assetPath, float ppu, SpriteAlignment alignment)
    {
        string absolutePath = ToAbsoluteAssetPath(assetPath);
        if (!File.Exists(absolutePath))
        {
            return;
        }

        AssetDatabase.ImportAsset(assetPath, ImportAssetOptions.ForceUpdate);
        TextureImporter importer = AssetImporter.GetAtPath(assetPath) as TextureImporter;
        if (importer == null)
        {
            return;
        }

        importer.textureType = TextureImporterType.Sprite;
        importer.spriteImportMode = SpriteImportMode.Single;
        importer.spritePixelsPerUnit = ppu;
        importer.filterMode = FilterMode.Bilinear;
        importer.textureCompression = TextureImporterCompression.Uncompressed;
        importer.alphaIsTransparency = false;

        var settings = new TextureImporterSettings();
        importer.ReadTextureSettings(settings);
        settings.spriteAlignment = (int)alignment;
        settings.spritePivot = new Vector2(0.5f, 0.5f);
        importer.SetTextureSettings(settings);
        importer.SaveAndReimport();
    }

    private static void BuildStageScene()
    {
        var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);

        GameObject systemRoot = CreateRoot("00_System");
        GameObject floorRoot = CreateRoot("01_Floor");
        GameObject backgroundRoot = CreateRoot("02_Background");
        GameObject obstacleRoot = CreateRoot("03_Obstacles");
        GameObject decalRoot = CreateRoot("04_Decals");
        GameObject itemRoot = CreateRoot("05_Items");
        GameObject enemyRoot = CreateRoot("06_Enemies");
        GameObject triggerRoot = CreateRoot("07_Triggers");
        GameObject uiRoot = CreateRoot("08_UI");
        GameObject cameraRoot = CreateRoot("09_Camera");

        PlacePrefab(GridPrefabPath, Vector3.zero, systemRoot.transform, "GridManager");
        PlacePrefab(GameStatePrefabPath, Vector3.zero, systemRoot.transform, "GameState");

        GameObject player = PlacePrefab(PlayerPrefabPath, new Vector3(52f, 5f, 0f), null, "Player");
        ConfigurePlayerSceneInstance(player);

        Camera cam = EnsureCamera(cameraRoot.transform);
        CameraFollow cameraFollow = EnsureComponent<CameraFollow>(cam.gameObject);
        cameraFollow.target = player.transform;
        cameraFollow.smoothSpeed = 8f;
        cameraFollow.offset = new Vector3(0f, 1.2f, -10f);
        cameraFollow.useClamp = true;
        cameraFollow.clampToCameraView = true;
        cameraFollow.minPosition = new Vector2(0.5f, 0.5f);
        cameraFollow.maxPosition = new Vector2(MapWidth - 0.5f, MapHeight - 0.5f);

        CreateGroundTiles(floorRoot.transform);
        CreateBackground(backgroundRoot.transform);
        CreateBoundaryTiles(obstacleRoot.transform);
        CreateHospitalFront(obstacleRoot.transform);
        CreateObstacleSet(obstacleRoot.transform);
        CreateDecalSet(decalRoot.transform);
        CreateItemSet(itemRoot.transform);
        CreateEnemySet(enemyRoot.transform, triggerRoot.transform);
        CreateHospitalTrigger(triggerRoot.transform, uiRoot.transform, player);

        EditorSceneManager.SaveScene(scene, ScenePath);
    }

        }

        var scene = EditorSceneManager.OpenScene(EntryScenePath, OpenSceneMode.Single);
        EntryUI entryUI = UnityEngine.Object.FindObjectOfType<EntryUI>();
        Button startButton = GameObject.Find("StartButton")?.GetComponent<Button>();
        if (entryUI != null && startButton != null)
        {
            startButton.onClick = new Button.ButtonClickedEvent();
            UnityEventTools.AddPersistentListener(startButton.onClick, entryUI.OnStartClicked);
            EditorSceneManager.SaveScene(scene);
        }
    }

    private static void UpdateBuildSettings()
    {
        List<EditorBuildSettingsScene> scenes = EditorBuildSettings.scenes.ToList();
        string[] desiredOrder =
        {
            EntryScenePath,
            ScenePath,
            "Assets/Project/Scenes/Hospital.unity",
            "Assets/Project/Scenes/Boss01_Elevator.unity",
            "Assets/Project/Scenes/RooftopEndingWalk.unity",
            "Assets/Project/Scenes/EndingCredits.unity",
            "Assets/Project/Scenes/Stage01_ParkingLobby.unity",
            "Assets/Project/Scenes/Boss02_Rooftop.unity",
        };

        var map = scenes.ToDictionary(s => s.path, s => s);
        var rebuilt = new List<EditorBuildSettingsScene>();
        foreach (string path in desiredOrder)
        {
            bool enabled = path != "Assets/Project/Scenes/Boss02_Rooftop.unity";
            rebuilt.Add(map.ContainsKey(path) ? new EditorBuildSettingsScene(path, enabled) : new EditorBuildSettingsScene(path, enabled));
        }

        EditorBuildSettings.scenes = rebuilt.ToArray();
    }

    private static void ConfigurePlayerSceneInstance(GameObject player)
    {
        WireAttackHitbox(player);

        SpriteRenderer spriteRenderer = EnsureComponent<SpriteRenderer>(player);
        spriteRenderer.sortingLayerName = "Characters";
        spriteRenderer.sortingOrder = 0;
        spriteRenderer.color = Color.white;

        AudioSource audioSource = EnsureComponent<AudioSource>(player);
        audioSource.playOnAwake = false;
        audioSource.loop = false;
        audioSource.spatialBlend = 0f;

        OneStepOnDirectionKey footsteps = EnsureComponent<OneStepOnDirectionKey>(player);
        var serialized = new SerializedObject(footsteps);
        AudioClip clip = AssetDatabase.LoadAssetAtPath<AudioClip>($"{AudioRoot}/Stage01_Footstep.wav");
        SerializedProperty clips = serialized.FindProperty("footstepClips");
        clips.arraySize = clip != null ? 1 : 0;
        if (clip != null)
        {
            clips.GetArrayElementAtIndex(0).objectReferenceValue = clip;
        }

        serialized.FindProperty("minimumInterval").floatValue = 0.05f;
        serialized.FindProperty("holdIntervalMultiplierMin").floatValue = 2.2f;
        serialized.FindProperty("holdIntervalMultiplierMax").floatValue = 4f;
        serialized.ApplyModifiedPropertiesWithoutUndo();

        PlayerVisualAnimator2D visuals = EnsureComponent<PlayerVisualAnimator2D>(player);
        bool generatedWalkAllExists = File.Exists(ToAbsoluteAssetPath(GeneratedWalkAllPath));
        bool generatedHurtAllExists = File.Exists(ToAbsoluteAssetPath(GeneratedHurtAllPath));

        visuals.walkDown = LoadOrderedSprites($"{ArtRoot}/Player/PlayerWalkDown.png");
        visuals.walkUp = LoadOrderedSprites($"{ArtRoot}/Player/PlayerWalkUp.png");
        visuals.walkRight = generatedWalkAllExists
            ? LoadGridRowSprites(GeneratedWalkAllPath, 0)
            : LoadOrderedSprites($"{ArtRoot}/Player/PlayerWalkRight.png");
        visuals.walkLeft = null;

        visuals.idleDown = CreateIdlePairFromWalk(visuals.walkDown);
        visuals.idleUp = CreateIdlePairFromWalk(visuals.walkUp);
        visuals.idleSide = CreateIdlePairFromWalk(visuals.walkRight);

        visuals.attackDown = LoadOrderedSprites(Stage01AttackDownPath);
        visuals.attackUp = LoadOrderedSprites(Stage01AttackUpPath);
        visuals.attackRight = LoadOrderedSprites(Stage01AttackSidePath);
        visuals.attackLeft = null;

        visuals.hurtDown = LoadGridRowSprites($"{ArtRoot}/Player/PlayerHurtAll.png", 0);
        visuals.hurtRight = generatedHurtAllExists
            ? LoadGridRowSprites(GeneratedHurtAllPath, 1)
            : LoadGridRowSprites($"{ArtRoot}/Player/PlayerHurtAll.png", 1);
        visuals.hurtUp = LoadGridRowSprites($"{ArtRoot}/Player/PlayerHurtAll.png", 2);
        visuals.hurtLeft = null;

        visuals.deadSequence = LoadOrderedSprites(Stage01DeadPath);

        if (visuals.idleDown != null && visuals.idleDown.Length > 0)
        {
            spriteRenderer.sprite = visuals.idleDown[0];
        }
    }

    {

    private static AnimationClip BuildLoopingSpriteClip(string assetPath, Sprite[] sprites, float framesPerSecond)
    {
        AnimationClip clip = AssetDatabase.LoadAssetAtPath<AnimationClip>(assetPath);
        if (clip == null)
        {
            clip = new AnimationClip();
            AssetDatabase.CreateAsset(clip, assetPath);
        }

        clip.frameRate = framesPerSecond;
        var binding = new EditorCurveBinding
        {
            type = typeof(SpriteRenderer),
            path = string.Empty,
            propertyName = "m_Sprite"
        };

        ObjectReferenceKeyframe[] keys = new ObjectReferenceKeyframe[sprites.Length];
        for (int i = 0; i < sprites.Length; i++)
        {
            keys[i] = new ObjectReferenceKeyframe
            {
                time = i / framesPerSecond,
                value = sprites[i]
            };
        }

        AnimationUtility.SetObjectReferenceCurve(clip, binding, keys);
        var settings = AnimationUtility.GetAnimationClipSettings(clip);
        settings.loopTime = true;
        AnimationUtility.SetAnimationClipSettings(clip, settings);
        EditorUtility.SetDirty(clip);
        return clip;
    }

    private static AnimatorController BuildController(string assetPath, Motion defaultMotion)
    {
        AnimatorController controller = AssetDatabase.LoadAssetAtPath<AnimatorController>(assetPath);
        if (controller == null)
        {
            controller = AnimatorController.CreateAnimatorControllerAtPath(assetPath);
        }

        AnimatorStateMachine stateMachine = controller.layers[0].stateMachine;
        AnimatorState state = stateMachine.states.Length > 0 ? stateMachine.states[0].state : null;
        if (state == null)
        {
            state = stateMachine.AddState("Walk");
        }

        state.motion = defaultMotion;
        stateMachine.defaultState = state;
        EditorUtility.SetDirty(controller);
        return controller;
    }

    private static void CopySprite(string sourceRelativePath, string destinationAssetPath, float ppu, SpriteAlignment alignment)
    {
        CopyFileFromDesktop(sourceRelativePath, destinationAssetPath);
        TextureImporter importer = (TextureImporter)AssetImporter.GetAtPath(destinationAssetPath);
        importer.textureType = TextureImporterType.Sprite;
        importer.spriteImportMode = SpriteImportMode.Single;
        importer.spritePixelsPerUnit = ppu;
        importer.filterMode = FilterMode.Point;
        importer.textureCompression = TextureImporterCompression.Uncompressed;
        importer.alphaIsTransparency = true;

        var settings = new TextureImporterSettings();
        importer.ReadTextureSettings(settings);
        settings.spriteAlignment = (int)alignment;
        settings.spritePivot = alignment == SpriteAlignment.BottomCenter ? new Vector2(0.5f, 0f) : new Vector2(0.5f, 0.5f);
        importer.SetTextureSettings(settings);
        importer.SaveAndReimport();
    }

    private static void CopyHorizontalSpriteSheet(string sourceRelativePath, string destinationAssetPath, int columns, float ppu, SpriteAlignment alignment)
    {
        CopyFileFromDesktop(sourceRelativePath, destinationAssetPath);
        TextureImporter importer = (TextureImporter)AssetImporter.GetAtPath(destinationAssetPath);
        Texture2D temp = LoadTextureFromFile(Path.Combine(SourceRoot, sourceRelativePath));
        int frameWidth = temp.width / columns;
        int frameHeight = temp.height;

        importer.textureType = TextureImporterType.Sprite;
        importer.spriteImportMode = SpriteImportMode.Multiple;
        importer.spritePixelsPerUnit = ppu;
        importer.filterMode = FilterMode.Point;
        importer.textureCompression = TextureImporterCompression.Uncompressed;
        importer.alphaIsTransparency = true;
        importer.spritesheet = Enumerable.Range(0, columns)
            .Select(index => new SpriteMetaData
            {
                alignment = (int)alignment,
                pivot = alignment == SpriteAlignment.BottomCenter ? new Vector2(0.5f, 0f) : new Vector2(0.5f, 0.5f),
                name = $"{Path.GetFileNameWithoutExtension(destinationAssetPath)}_{index + 1:D2}",
                rect = new Rect(index * frameWidth, 0f, frameWidth, frameHeight)
            })
            .ToArray();
        importer.SaveAndReimport();
        UnityEngine.Object.DestroyImmediate(temp);
    }

    private static void CopyGridSpriteSheet(string sourceRelativePath, string destinationAssetPath, int columns, int rows, float ppu, SpriteAlignment alignment)
    {
        CopyFileFromDesktop(sourceRelativePath, destinationAssetPath);
        TextureImporter importer = (TextureImporter)AssetImporter.GetAtPath(destinationAssetPath);
        Texture2D temp = LoadTextureFromFile(Path.Combine(SourceRoot, sourceRelativePath));
        int frameWidth = temp.width / columns;
        int frameHeight = temp.height / rows;
        int textureHeight = temp.height;

        importer.textureType = TextureImporterType.Sprite;
        importer.spriteImportMode = SpriteImportMode.Multiple;
        importer.spritePixelsPerUnit = ppu;
        importer.filterMode = FilterMode.Point;
        importer.textureCompression = TextureImporterCompression.Uncompressed;
        importer.alphaIsTransparency = true;

        var sprites = new List<SpriteMetaData>();
        for (int row = 0; row < rows; row++)
        {
            for (int col = 0; col < columns; col++)
            {
                sprites.Add(new SpriteMetaData
                {
                    alignment = (int)alignment,
                    pivot = alignment == SpriteAlignment.BottomCenter ? new Vector2(0.5f, 0f) : new Vector2(0.5f, 0.5f),
                    name = $"{Path.GetFileNameWithoutExtension(destinationAssetPath)}_r{row + 1:D2}_c{col + 1:D2}",
                    rect = new Rect(col * frameWidth, textureHeight - ((row + 1) * frameHeight), frameWidth, frameHeight)
                });
            }
        }

        importer.spritesheet = sprites.ToArray();
        importer.SaveAndReimport();
        UnityEngine.Object.DestroyImmediate(temp);
    }

    private static Texture2D RemoveNearWhiteBackground(Texture2D source, float threshold)
    {
        Texture2D result = new Texture2D(source.width, source.height, TextureFormat.RGBA32, false);
        Color[] pixels = source.GetPixels();

        for (int i = 0; i < pixels.Length; i++)
        {
            Color pixel = pixels[i];
            bool nearWhite = pixel.r >= threshold && pixel.g >= threshold && pixel.b >= threshold;
            if (nearWhite)
            {
                pixel.a = 0f;
            }

            pixels[i] = pixel;
        }

        result.SetPixels(pixels);
        result.Apply();
        return result;
    }

    private static void SaveTextureToAssetPath(Texture2D texture, string destinationAssetPath)
    {
        string destinationAbsolute = ToAbsoluteAssetPath(destinationAssetPath);

        AssetDatabase.ImportAsset(destinationAssetPath, ImportAssetOptions.ForceUpdate);
    }

    private static void WireAttackHitbox(GameObject player)
    {
        PlayerCombat combat = player.GetComponent<PlayerCombat>();
        if (combat == null)
        {
            return;
        }

        combat.hitboxFlashPrefab = null;
        combat.drawAttackGizmo = false;
    }

    private static Sprite GetAsphaltTile(int x, int y)
    {
        string[] assetNames =
        {
            "Asphalt01",
            "Asphalt02",
            "Asphalt03",
            "Asphalt04",
            "Asphalt05",
            "Asphalt06",
        };

        string selected = assetNames[Mathf.Abs((x * 13 + y * 7) % assetNames.Length)];
        return AssetDatabase.LoadAssetAtPath<Sprite>($"{ArtRoot}/Tiles/{selected}.png");
    }

    private static Sprite GetCrackTile(int x, int y)
    {
        string assetName = ((x + y) % 2 == 0) ? "AsphaltCrack01" : "AsphaltCrack04";
        return AssetDatabase.LoadAssetAtPath<Sprite>($"{ArtRoot}/Tiles/{assetName}.png");
    }
